### 使用示例

AI 翻译挂件（vcpAPI.post 调用 AI）：
<<<[DESKTOP_PUSH]>>>
<div>
  <textarea id="src" placeholder="输入文本..."></textarea>
  <button id="btn">翻译为英文</button>
  <div id="out"></div>
</div>
<script>
document.getElementById('btn').addEventListener('click', async function() {
    var text = document.getElementById('src').value;
    if (!text) return;
    document.getElementById('out').textContent = '翻译中...';
    var result = await vcpAPI.post([
        { role: "system", content: "你是翻译引擎，只输出翻译结果。" },
        { role: "user", content: "翻译为英文：\n\n" + text }
    ], { model: "gemini-3-flash-preview", temperature: 0.3 });
    document.getElementById('out').textContent = result.success ? result.content : '错误: ' + result.error;
});
</script>
<<<[DESKTOP_PUSH_END]>>>

音乐控制条（musicAPI 控制播放）：
<<<[DESKTOP_PUSH]>>>
<div>
  <span id="title">--</span>
  <button onclick="musicAPI.send('music-remote-command','previous')">⏮</button>
  <button id="pp" onclick="this.dataset.playing==='true'?musicAPI.pause():musicAPI.play()">▶</button>
  <button onclick="musicAPI.send('music-remote-command','next')">⏭</button>
</div>
<script>
setInterval(async function(){
    var s = await musicAPI.getState();
    if(!s) return;
    var pp = document.getElementById('pp');
    pp.textContent = s.is_playing ? '⏸' : '▶';
    pp.dataset.playing = s.is_playing;
    var fp = s.file_path || '';
    var parts = fp.split(/[\\\\/]/);
    var name = parts[parts.length-1] || '--';
    var dot = name.lastIndexOf('.');
    if(dot>0) name = name.substring(0,dot);
    document.getElementById('title').textContent = name;
}, 2000);
</script>
<<<[DESKTOP_PUSH_END]>>>

var currentPath = ".";
var curDirEl = document.getElementById("cur-dir");
var fileListEl = document.getElementById("file-list");

async function loadDir(path) {
    currentPath = path;
    curDirEl.textContent = "当前目录: " + path;
    fileListEl.innerHTML = "<div style='color:#6c7086'>加载中...</div>";
    try {
        var res = await nfm.list(path);
        fileListEl.innerHTML = "";
        if (res && res.entries) {
            res.entries.forEach(function(item) {
                var div = document.createElement("div");
                div.style.cssText = "padding:6px; cursor:pointer; border-bottom:1px solid #313244; display:flex; justify-content:space-between; align-items:center;";
                
                var nameSpan = document.createElement("span");
                nameSpan.textContent = (item.isDir ? "📁 " : "📄 ") + item.name;
                div.appendChild(nameSpan);
                
                var actionsSpan = document.createElement("span");
                actionsSpan.style.cssText = "display:flex; gap:8px;";
                
                var btnCut = document.createElement("button");
                btnCut.textContent = "剪切";
                btnCut.onclick = function(e) {
                    e.stopPropagation();
                    nfm.cutToClipboard(item.path).then(function() { alert("已剪切: " + item.name); });
                };
                actionsSpan.appendChild(btnCut);
                
                var btnDel = document.createElement("button");
                btnDel.textContent = "删除";
                btnDel.style.color = "#f38ba8";
                btnDel.onclick = function(e) {
                    e.stopPropagation();
                    if(confirm("确定删除 " + item.name + " 吗？")) {
                        nfm.trash(item.path).then(function() { loadDir(currentPath); });
                    }
                };
                actionsSpan.appendChild(btnDel);
                
                div.appendChild(actionsSpan);
                
                div.onclick = function() {
                    if (item.isDir) {
                        loadDir(item.path);
                    } else {
                        nfm.open(item.path);
                    }
                };
                fileListEl.appendChild(div);
            });
        }
    } catch(err) {
        fileListEl.innerHTML = "<div style='color:#f38ba8'>加载出错: " + err + "</div>";
    }
}

document.getElementById("btn-back").onclick = function() {
    if (currentPath === "." || currentPath === "") return;
    var parts = currentPath.split("/");
    parts.pop();
    loadDir(parts.join("/") || ".");
};

document.getElementById("btn-new").onclick = function() {
    var name = prompt("请输入文件夹名称:");
    if (name) {
        nfm.newFolder(currentPath, name).then(function() { loadDir(currentPath); });
    }
};

document.getElementById("btn-paste").onclick = function() {
    nfm.paste(currentPath, "rename").then(function(r) {
        if(r.success) {
            loadDir(currentPath);
        } else {
            alert("粘贴失败: " + r.error);
        }
    });
};

nfm.onFileChange(function(data) {
    if (data.directoryPath === currentPath) {
        loadDir(currentPath);
    }
});

loadDir(".");
